## IDENTITAS PROYEK

Kamu adalah senior full-stack developer berpengalaman. Bangun website **"Sistem Informasi Pelestarian Warisan Budaya Kebendaan Suku Sasak"** yang berjalan 100% di localhost tanpa deployment.

Website ini adalah hasil penelitian lapangan tentang 10 warisan budaya kebendaan (tangible heritage) di Lombok Timur, NTB. Ada dua jenis pengguna: **Admin** (login, kelola data) dan **Pengunjung** (lihat saja).

```php
<?php
namespace App\Models;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Str;

class HeritageSite extends Model
{
    protected $table = 'heritage_sites';
    
    protected $fillable = [
        'nama', 'slug', 'kategori', 'lat', 'lng',
        'deskripsi', 'sejarah', 'tahun_berdiri',
        'wilayah', 'kecamatan', 'kondisi_saat_ini',
        'skor_kondisi_cnn', 'status_aktif'
    ];

    protected $casts = [
        'lat' => 'float',
        'lng' => 'float',
        'skor_kondisi_cnn' => 'float',
        'status_aktif' => 'boolean',
    ];

    protected static function boot()
    {
        parent::boot();
        static::creating(function ($model) {
            if (empty($model->slug)) {
                $model->slug = Str::slug($model->nama);
            }
        });
    }

    public function photos()
    {
        return $this->hasMany(HeritagePhoto::class, 'heritage_id')->orderBy('urutan');
    }

    public function primaryPhoto()
    {
        return $this->hasOne(HeritagePhoto::class, 'heritage_id')->where('is_primary', true);
    }

    public function scopeKategori($query, $kategori)
    {
        return $query->where('kategori', $kategori);
    }

    public function scopeWilayah($query, $wilayah)
    {
        return $query->where('wilayah', $wilayah);
    }

    public function toGeoJsonFeature(): array
    {
        return [
            'type' => 'Feature',
            'geometry' => [
                'type' => 'Point',
                'coordinates' => [$this->lng, $this->lat]
            ],
            'properties' => [
                'id' => $this->id,
                'nama' => $this->nama,
                'kategori' => $this->kategori,
                'wilayah' => $this->wilayah,
                'kondisi' => $this->kondisi_saat_ini,
                'thumbnail' => $this->primaryPhoto?->url_thumbnail,
            ]
        ];
    }
}
```

```php
<?php
namespace App\Http\Controllers\Api;
use App\Http\Controllers\Controller;
use App\Models\HeritageSite;
use Illuminate\Http\Request;

class HeritageController extends Controller
{
    // GET /api/heritage — publik, dengan filter & search
    public function index(Request $request)
    {
        $query = HeritageSite::with(['primaryPhoto'])->where('status_aktif', true);

        if ($request->has('kategori') && $request->kategori !== '') {
            $query->where('kategori', $request->kategori);
        }
        if ($request->has('wilayah') && $request->wilayah !== '') {
            $query->where('wilayah', $request->wilayah);
        }
        if ($request->has('q') && $request->q !== '') {
            $query->where(function($q) use ($request) {
                $q->where('nama', 'like', "%{$request->q}%")
                  ->orWhere('deskripsi', 'like', "%{$request->q}%")
                  ->orWhere('wilayah', 'like', "%{$request->q}%");
            });
        }

        $data = $query->get()->map(fn($h) => [
            'id' => $h->id,
            'nama' => $h->nama,
            'slug' => $h->slug,
            'kategori' => $h->kategori,
            'wilayah' => $h->wilayah,
            'tahun_berdiri' => $h->tahun_berdiri,
            'kondisi_saat_ini' => $h->kondisi_saat_ini,
            'skor_kondisi_cnn' => $h->skor_kondisi_cnn,
            'lat' => $h->lat,
            'lng' => $h->lng,
            'thumbnail' => $h->primaryPhoto?->url_thumbnail,
        ]);

        return response()->json(['data' => $data, 'total' => $data->count()]);
    }

    // GET /api/heritage/{id} — detail lengkap
    public function show($id)
    {
        $heritage = HeritageSite::with('photos')->findOrFail($id);
        return response()->json([
            'data' => array_merge($heritage->toArray(), [
                'foto' => $heritage->photos->map(fn($p) => [
                    'id' => $p->id,
                    'url_foto' => asset('storage/' . $p->url_foto),
                    'url_thumbnail' => $p->url_thumbnail ? asset('storage/' . $p->url_thumbnail) : null,
                    'label_kondisi_cnn' => $p->label_kondisi_cnn,
                    'skor_cnn' => $p->skor_cnn,
                    'is_primary' => $p->is_primary,
                ])
            ])
        ]);
    }

    // GET /api/heritage/geojson — untuk Leaflet
    public function geojson()
    {
        $features = HeritageSite::with('primaryPhoto')
            ->where('status_aktif', true)
            ->get()
            ->map(fn($h) => $h->toGeoJsonFeature());

        return response()->json([
            'type' => 'FeatureCollection',
            'features' => $features
        ]);
    }

    // POST /api/admin/heritage — admin tambah (protected)
    public function store(Request $request)
    {
        $validated = $request->validate([
            'nama'         => 'required|string|max:200',
            'kategori'     => 'required|in:masjid,makam,rumah_adat,desa_adat',
            'lat'          => 'required|numeric|between:-9.5,-8.0',
            'lng'          => 'required|numeric|between:115.5,117.5',
            'wilayah'      => 'required|string',
            'deskripsi'    => 'nullable|string',
            'sejarah'      => 'nullable|string',
            'tahun_berdiri'=> 'nullable|string',
            'kecamatan'    => 'nullable|string',
            'kondisi_saat_ini' => 'nullable|in:baik,sedang,buruk,belum_dikaji',
        ]);

        $heritage = HeritageSite::create($validated);
        return response()->json(['data' => $heritage, 'message' => 'Heritage berhasil ditambahkan'], 201);
    }

    // PUT /api/admin/heritage/{id} — admin edit (protected)
    public function update(Request $request, $id)
    {
        $heritage = HeritageSite::findOrFail($id);
        $heritage->update($request->validate([
            'nama'         => 'sometimes|string|max:200',
            'kategori'     => 'sometimes|in:masjid,makam,rumah_adat,desa_adat',
            'lat'          => 'sometimes|numeric|between:-9.5,-8.0',
            'lng'          => 'sometimes|numeric|between:115.5,117.5',
            'wilayah'      => 'sometimes|string',
            'deskripsi'    => 'nullable|string',
            'sejarah'      => 'nullable|string',
            'tahun_berdiri'=> 'nullable|string',
            'kondisi_saat_ini' => 'nullable|in:baik,sedang,buruk,belum_dikaji',
            'status_aktif' => 'sometimes|boolean',
        ]));
        return response()->json(['data' => $heritage, 'message' => 'Heritage berhasil diperbarui']);
    }

    // DELETE /api/admin/heritage/{id} — admin hapus (protected)
    public function destroy($id)
    {
        HeritageSite::findOrFail($id)->delete();
        return response()->json(['message' => 'Heritage berhasil dihapus']);
    }
}
```

**app/Http/Controllers/Api/PhotoController.php**
```php
<?php
namespace App\Http\Controllers\Api;
use App\Http\Controllers\Controller;
use App\Models\HeritagePhoto;
use App\Models\HeritageSite;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Storage;
use Intervention\Image\Facades\Image;

class PhotoController extends Controller
{
    public function store(Request $request, $heritageId)
    {
        $request->validate([
            'foto' => 'required|file|mimes:jpeg,jpg,png,webp|max:10240',
        ]);

        $heritage = HeritageSite::findOrFail($heritageId);
        $file = $request->file('foto');

        // Simpan foto original
        $path = $file->store('heritage-photos', 'public');

        // Buat thumbnail (simpan versi kecil)
        $thumbPath = 'heritage-photos/thumb_' . basename($path);
        // Jika tidak ada Intervention Image, skip thumbnail
        $thumbPathFull = null;
        try {
            $img = Image::make($file)->fit(400, 300);
            Storage::disk('public')->put($thumbPath, $img->encode());
            $thumbPathFull = $thumbPath;
        } catch (\Throwable $e) {
            $thumbPathFull = $path; // fallback ke foto asli
        }

        // Analisis CNN (synchronous untuk localhost — sederhana)
        $cnnLabel = 'belum_dianalisis';
        $cnnSkor = null;
        try {
            $cnnUrl = env('CNN_SERVICE_URL', 'http://localhost:5000');
            $response = Http::timeout(15)->attach(
                'foto',
                Storage::disk('public')->get($path),
                basename($path)
            )->post("$cnnUrl/predict");

            if ($response->successful()) {
                $result = $response->json();
                $cnnLabel = $result['label'] ?? 'belum_dianalisis';
                $cnnSkor = $result['confidence'] ?? null;
            }
        } catch (\Throwable $e) {
            // CNN tidak wajib jalan — lanjut tanpa error
        }

        // Cek apakah ini foto pertama (jadi primary otomatis)
        $isPrimary = $heritage->photos()->count() === 0;

        $foto = HeritagePhoto::create([
            'heritage_id'       => $heritageId,
            'url_foto'          => $path,
            'url_thumbnail'     => $thumbPathFull,
            'label_kondisi_cnn' => $cnnLabel,
            'skor_cnn'          => $cnnSkor,
            'is_primary'        => $isPrimary,
            'urutan'            => $heritage->photos()->count(),
        ]);

        // Update kondisi heritage dari hasil CNN
        if ($cnnLabel !== 'belum_dianalisis') {
            $heritage->update([
                'kondisi_saat_ini' => $cnnLabel,
                'skor_kondisi_cnn' => $cnnSkor,
            ]);
        }

        return response()->json([
            'data' => [
                'id' => $foto->id,
                'url_foto' => asset('storage/' . $path),
                'url_thumbnail' => asset('storage/' . $thumbPathFull),
                'label_kondisi_cnn' => $cnnLabel,
                'skor_cnn' => $cnnSkor,
                'is_primary' => $isPrimary,
            ],
            'message' => 'Foto berhasil diupload' . ($cnnLabel !== 'belum_dianalisis' ? " — CNN: $cnnLabel" : ' — CNN belum dianalisis')
        ], 201);
    }

    public function destroy($heritageId, $fotoId)
    {
        $foto = HeritagePhoto::where('heritage_id', $heritageId)->findOrFail($fotoId);
        Storage::disk('public')->delete($foto->url_foto);
        if ($foto->url_thumbnail && $foto->url_thumbnail !== $foto->url_foto) {
            Storage::disk('public')->delete($foto->url_thumbnail);
        }
        $foto->delete();
        return response()->json(['message' => 'Foto berhasil dihapus']);
    }

    public function setPrimary($heritageId, $fotoId)
    {
        HeritagePhoto::where('heritage_id', $heritageId)->update(['is_primary' => false]);
        HeritagePhoto::where('heritage_id', $heritageId)->findOrFail($fotoId)->update(['is_primary' => true]);
        return response()->json(['message' => 'Foto utama berhasil diubah']);
    }
}
```

**app/Http/Controllers/Api/AuthController.php**
```php
<?php
namespace App\Http\Controllers\Api;
use App\Http\Controllers\Controller;
use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\RateLimiter;

class AuthController extends Controller
{
    public function login(Request $request)
    {
        $request->validate([
            'email'    => 'required|email',
            'password' => 'required',
        ]);

        // Rate limiting: 5 percobaan per 15 menit
        $key = 'login:' . $request->ip();
        if (RateLimiter::tooManyAttempts($key, 5)) {
            $seconds = RateLimiter::availableIn($key);
            return response()->json([
                'message' => "Terlalu banyak percobaan login. Coba lagi dalam {$seconds} detik."
            ], 429);
        }

        if (!Auth::attempt($request->only('email', 'password'))) {
            RateLimiter::hit($key, 60 * 15);
            return response()->json(['message' => 'Email atau password salah.'], 401);
        }

        RateLimiter::clear($key);
        $user = Auth::user();
        $token = $user->createToken('admin-token')->plainTextToken;

        return response()->json([
            'token' => $token,
            'user'  => ['id' => $user->id, 'nama' => $user->name, 'email' => $user->email],
            'message' => 'Login berhasil'
        ]);
    }

    public function logout(Request $request)
    {
        $request->user()->currentAccessToken()->delete();
        return response()->json(['message' => 'Logout berhasil']);
    }

    public function me(Request $request)
    {
        return response()->json(['user' => $request->user()]);
    }
}
```

```js
import axios from 'axios';

const api = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL + '/api',
  headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' }
});

// Pasang token otomatis ke semua request
api.interceptors.request.use(config => {
  const token = localStorage.getItem('admin_token');
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

// Auto redirect jika 401
api.interceptors.response.use(
  res => res,
  err => {
    if (err.response?.status === 401) {
      localStorage.removeItem('admin_token');
      window.location.href = '/login';
    }
    return Promise.reject(err);
  }
);

export const heritageApi = {
  getAll: (params) => api.get('/heritage', { params }),
  getById: (id) => api.get(`/heritage/${id}`),
  getGeoJson: () => api.get('/heritage/geojson'),
  create: (data) => api.post('/admin/heritage', data),
  update: (id, data) => api.put(`/admin/heritage/${id}`, data),
  delete: (id) => api.delete(`/admin/heritage/${id}`),
  uploadPhoto: (heritageId, formData) => api.post(
    `/admin/heritage/${heritageId}/photos`, formData,
    { headers: { 'Content-Type': 'multipart/form-data' } }
  ),
  deletePhoto: (heritageId, fotoId) => api.delete(`/admin/heritage/${heritageId}/photos/${fotoId}`),
};

export const authApi = {
  login: (email, password) => api.post('/auth/login', { email, password }),
  logout: () => api.post('/auth/logout'),
  me: () => api.get('/auth/me'),
};

export default api;
```

### 3.2 cnn-service/app.py
```python
from flask import Flask, request, jsonify
import numpy as np
import os
import io
from PIL import Image

app = Flask(__name__)

# Load model saat startup
model = None
CLASS_LABELS = ['baik', 'sedang', 'buruk']

def load_model():
    global model
    model_path = os.path.join(os.path.dirname(__file__), 'model', 'heritage_cnn.h5')
    if os.path.exists(model_path):
        import tensorflow as tf
        model = tf.keras.models.load_model(model_path)
        print(f"✅ Model CNN berhasil dimuat dari {model_path}")
    else:
        print("⚠️  Model CNN tidak ditemukan — endpoint /predict akan return dummy data")

def preprocess_image(image_bytes):
    img = Image.open(io.BytesIO(image_bytes)).convert('RGB')
    img = img.resize((224, 224))
    arr = np.array(img, dtype=np.float32) / 255.0
    # Normalize ImageNet
    mean = np.array([0.485, 0.456, 0.406])
    std  = np.array([0.229, 0.224, 0.225])
    arr = (arr - mean) / std
    return np.expand_dims(arr, axis=0)

@app.route('/health', methods=['GET'])
def health():
    return jsonify({
        'status': 'ok',
        'model_loaded': model is not None,
        'message': 'CNN Service berjalan'
    })

@app.route('/predict', methods=['POST'])
def predict():
    if 'foto' not in request.files:
        return jsonify({'error': 'File foto tidak ditemukan'}), 400

    file = request.files['foto']
    if file.filename == '':
        return jsonify({'error': 'Nama file kosong'}), 400

    try:
        image_bytes = file.read()

        # Jika model belum dimuat, return dummy data
        if model is None:
            # Dummy: acak untuk testing
            import random
            label = random.choice(CLASS_LABELS)
            skor = round(random.uniform(0.6, 0.95), 4)
            return jsonify({
                'label': label,
                'confidence': skor,
                'skor_baik':   round(random.uniform(0.0, 0.3), 4),
                'skor_sedang': round(random.uniform(0.0, 0.3), 4),
                'skor_buruk':  round(random.uniform(0.0, 0.3), 4),
                'model_version': 'dummy-v0',
                'perlu_verifikasi': skor < 0.60,
                'catatan': 'Model belum dimuat — hasil ini adalah dummy untuk testing'
            })

        # Proses dengan model sungguhan
        arr = preprocess_image(image_bytes)
        predictions = model.predict(arr)[0]
        idx = int(np.argmax(predictions))
        label = CLASS_LABELS[idx]
        confidence = float(predictions[idx])

        return jsonify({
            'label': label,
            'confidence': round(confidence, 4),
            'skor_baik':   round(float(predictions[0]), 4),
            'skor_sedang': round(float(predictions[1]), 4),
            'skor_buruk':  round(float(predictions[2]), 4),
            'model_version': 'v1.0',
            'perlu_verifikasi': confidence < 0.60
        })

    except Exception as e:
        return jsonify({'error': str(e), 'label': None}), 500

if __name__ == '__main__':
    load_model()
    print("🚀 CNN Service berjalan di http://localhost:5000")
    app.run(host='0.0.0.0', port=5000, debug=True)
```

> **Catatan:** CNN akan jalan dengan dummy data acak jika file model `.h5` belum ada. Ini memungkinkan testing website tanpa harus training CNN dulu. Upload foto tetap berhasil, label CNN akan berupa nilai acak sementara.
